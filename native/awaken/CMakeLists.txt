cmake_minimum_required(VERSION 3.20)
project(bcm_awaken LANGUAGES C CXX ASM)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Awaken bridge requires Windows x64")
endif()
add_subdirectory(../external/vendor/minhook ${CMAKE_CURRENT_BINARY_DIR}/minhook)
add_library(bcm_awaken SHARED bridge.cpp hooks.cpp tick.S)
target_compile_features(bcm_awaken PRIVATE cxx_std_17)
target_compile_options(bcm_awaken PRIVATE -Wall -Wextra -Werror)
target_link_options(bcm_awaken PRIVATE -static)
target_link_libraries(bcm_awaken PRIVATE advapi32 ole32 bcrypt minhook)
set_target_properties(bcm_awaken PROPERTIES PREFIX "" OUTPUT_NAME "bcm-awaken")
add_executable(test_awaken_tick test_tick.cpp tick.S)
target_compile_features(test_awaken_tick PRIVATE cxx_std_17)
target_link_options(test_awaken_tick PRIVATE -static)
enable_testing()
add_test(NAME awaken_tick_abi COMMAND test_awaken_tick)
add_executable(test_startup_detour test_startup_detour.cpp)
target_compile_features(test_startup_detour PRIVATE cxx_std_17)
target_link_options(test_startup_detour PRIVATE -static)
add_test(NAME awaken_startup_detour COMMAND test_startup_detour)

add_executable(test_engine_chain test_engine_chain.cpp)
target_compile_features(test_engine_chain PRIVATE cxx_std_17)
target_link_options(test_engine_chain PRIVATE -static)
add_test(NAME awaken_engine_chain COMMAND test_engine_chain)
